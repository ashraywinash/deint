
Create the Hive Table

CREATE TABLE user_data (
  data_date STRING,
  user_id STRING,
  properties STRING
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY '\t'
STORED AS TEXTFILE;

Extract state from properties field
SELECT 
  regexp_extract(properties, 'state=([^;]+)', 1) AS state
FROM user_data;

Count per State
SELECT 
  regexp_extract(properties, 'state=([^;]+)', 1) AS state,
  COUNT(*) AS state_count
FROM user_data
GROUP BY regexp_extract(properties, 'state=([^;]+)', 1);

Minimum, Maximum Age, and Unique Count
SELECT
  MIN(CAST(regexp_extract(properties, 'Age=([0-9]+)', 1) AS INT)) AS min_age,
  MAX(CAST(regexp_extract(properties, 'Age=([0-9]+)', 1) AS INT)) AS max_age,
  COUNT(DISTINCT CAST(regexp_extract(properties, 'Age=([0-9]+)', 1) AS INT)) AS unique_age_count
FROM user_data;

Count Records per Gender
SELECT 
  regexp_extract(properties, 'gender=([MF])', 1) AS gender,
  COUNT(*) AS gender_count
FROM user_data
GROUP BY regexp_extract(properties, 'gender=([MF])', 1);
